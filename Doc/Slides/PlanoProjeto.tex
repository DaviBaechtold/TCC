\section{Objetivo e Escopo}
\begin{frame}{Objetivo}
  \begin{itemize}
    \item Desenvolver um pipeline geral de reconhecimento de gestos/ações a partir de vídeos RGB monoculares via esqueletos 2D (mãos/pose) e Transformer.
    \item Foco abrangente (HCI, XR, robótica, esportes, segurança, educação) – não restrito a comunicação entre surdos e mudos.
    \item Entrega: código reprodutível, modelos treinados, demo em tempo real e documentação.
  \end{itemize}
\end{frame}

\begin{frame}{Escopo}
  \begin{itemize}
    \item Extração de keypoints com MediaPipe (mãos/pose) e preparação de sequências temporais.
    \item Classificação com Transformer encoder; baselines LSTM/MLP para comparação.
    \item Treinamento, avaliação e demo em tempo real com webcam.
    \item Suporte a múltiplos datasets (ex.: Jester, EgoGesture, NTU RGB+D (2D), HaGRID, etc.).
  \end{itemize}
\end{frame}

\section{Pipeline Técnico}
\begin{frame}{Arquitetura do Pipeline}
  \begin{itemize}
    \item \textbf{Entrada}: vídeo RGB ou sequência de frames.
    \item \textbf{Extração}: MediaPipe Hands/Pose $\Rightarrow$ keypoints 2D por frame.
    \item \textbf{Pré-processamento}: pad/trim para $T$ quadros, normalização (z-score por amostra), máscara temporal.
  \item \textbf{Modelo}: projeção linear $\Rightarrow$ positional encoding $\Rightarrow$ Transformer encoder (multihead) $\Rightarrow$ pooling (CLS) $\Rightarrow$ cabeça MLP.
    \item \textbf{Saída}: logits $\Rightarrow$ classe de gesto/ação.
  \end{itemize}
\end{frame}

\section{Bases de Dados}
\begin{frame}{Datasets considerados}
  \begin{itemize}
    \item \textbf{20BN-Jester} (dinâmico, mãos): leve, ótimo para Gestos manuais; já integrado.
    \item \textbf{EgoGesture} (dinâmico, egocêntrico): variedade de gestos, cenário natural; maior porte.
    \item \textbf{HaGRID} (estático, mãos): 2.8M imagens; útil para pré-treino de detector/classificador estático.
    \item \textbf{NTU RGB+D 60/120} (ações corporais): pode-se extrair 2D (MediaPipe) para usar no pipeline.
    \item \textbf{ChaLearn LAP IsoGD / Montalbano} (gestos): conjuntos multimodais (RGB-D), uso acadêmico.
    \item \textbf{Something-Something V2 / Kinetics} (ações gerais): muito grandes; seleção de subset recomendada.
  \end{itemize}
\end{frame}

\begin{frame}{20BN-Jester (detalhes)}
  \begin{itemize}
    \item Dados: sequências de frames por vídeo (RGB), gestos manuais dinâmicos (ex.: deslizar, apontar, rodar).
    \item Rótulos: ação por vídeo (classe única).
    \item Integração: pasta de frames \Rightarrow MediaPipe Hands \Rightarrow `.npz` (sequência de keypoints) + manifesto CSV.
  \end{itemize}
\end{frame}

\begin{frame}{EgoGesture (detalhes)}
  \begin{itemize}
    \item Dados: vídeos egocêntricos com gestos manuais; condições variadas (fundo, iluminação).
    \item Rótulos: conjunto de gestos predefinidos; metadados por sujeito/sessão.
    \item Integração: vídeos \Rightarrow extração de frames \Rightarrow MediaPipe Hands/Pose \Rightarrow `.npz` + manifesto.
  \end{itemize}
\end{frame}

\begin{frame}{NTU RGB+D (detalhes)}
  \begin{itemize}
    \item Dados: ações corporais multi-sujeito; sensores RGB-D (usaremos apenas RGB \& 2D keypoints).
    \item Rótulos: ação por vídeo (multi-categoria); particionamento por sujeito/câmera.
    \item Integração: vídeos \Rightarrow MediaPipe Pose \Rightarrow `.npz` (pose 2D por frame) + manifesto.
  \end{itemize}
\end{frame}

\begin{frame}{Unificação de múltiplos datasets}
  \begin{itemize}
    \item Normalização de labels: mapear gestos similares para uma taxonomia comum (ex.: \emph{swipe-left/right}, \emph{pointing}).
    \item Manifesto unificado: `path,label` com one-hot via `class_to_idx` consistente entre bases.
    \item Balanceamento: amostragem estratificada por classe e por origem (dataset) para evitar viés.
    \item Validação cruzada: splits por vídeo/sujeito; manter splits separados por origem quando necessário.
  \end{itemize}
\end{frame}

\begin{frame}{Critérios de seleção}
  \begin{itemize}
    \item Licença/uso acadêmico, tamanho e facilidade de download.
    \item Alinhamento com os gestos-alvo e com o escopo geral.
    \item Facilidade de integração: frames em diretórios ou vídeos; rótulos claros; divisão treino/val/teste.
  \end{itemize}
\end{frame}

\section{Literature Review e Justificativas}
\begin{frame}{Literature Review (resumo)}
  \begin{itemize}
    \item \textbf{2D Skeleton-based}: ST-GCN, TCNs, RNN/LSTM; foco em topologia e dinâmica (citações a inserir).
    \item \textbf{Transformers} para séries temporais e vídeo: atenção global, modelagem de longo alcance (citações a inserir).
    \item \textbf{Modelos híbridos}: atenção espacial-temporal, pooling adaptativo, multi-modal (RGB + keypoints) (citações a inserir).
  \end{itemize}
\end{frame}

\begin{frame}{Por que 2D Skeletons?}
  \begin{itemize}
    \item Robustez a variações de aparência, iluminação e fundo; menor custo de processamento que vídeo bruto.
    \item Privacidade: reduz exposição de imagem completa ao trabalhar com coordenadas.
    \item Compatível com câmeras simples (RGB) e bibliotecas como MediaPipe (tempo real em CPU).
  \end{itemize}
\end{frame}

\begin{frame}{Por que Transformers?}
  \begin{itemize}
    \item Captura dependências de longo alcance via auto-atenção; bom para gestos com duração variável.
    \item Flexível para incorporar tokens especiais (CLS), máscaras temporais e multi-head attention.
    \item Resultados competitivos em visão e séries temporais; boa base para extensões.
  \end{itemize}
\end{frame}
